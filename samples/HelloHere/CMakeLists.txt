# Пример приложения -- так выглядит программа, написанная на wxl: она линкует
# wxl::ui и никогда не видит проекцию cppwinrt.

# wxl_add_executable() -- это всё, что для этого нужно: исполняемый файл
# подсистемы GUI, чтобы точкой входа стал собственный wWinMain библиотеки wxl,
# слинкованный с wxl::ui и с тем, что нужно библиотеке во время работы.
wxl_add_executable(sample.hello
    main.cpp
)

# Еще один хелпер - иконка едет внутри исполняемого файла: её показывает Проводник,
# её же wxl ставит окну автоматически.
wxl_target_icon(sample.hello app.ico)

# Uri ресурса без схемы ведёт к пути рядом с исполняемым файлом, поэтому
# картинки должны лежать там же. Каждый файл папки -- выход сборки со своей
# зависимостью, поэтому правка одного файла копирует только его.
wxl_target_assets(sample.hello Assets)
